## §4 修法說明

### §4.1 `web_server.py` — 新增 `GET /api/themes`

在 `upload_theme` 函式結尾（`return {...}` 後）插入 `list_themes()`：

```python
@app.get("/api/themes")
async def list_themes():
    """RAG-13：取得所有內建與上傳的自訂主題。

    1. 內建主題：mies, kahn, kandinsky, nara（custom=False, updated_at=0.0）
    2. 自訂主題：讀取 static/themes/ 目錄下非內建的 .css 檔案，附 st_mtime
    3. 排除內建的主題檔名，避免重複
    """
    themes_dir = BASE_DIR / "static" / "themes"
    built_in = {
        "mies": "Mies van der Rohe",
        "kahn": "Kahn · Kimbell Art Museum",
        "kandinsky": "Kandinsky · Bauhaus",
        "nara": "Yoshitomo Nara",
    }

    result = []

    # 1. 加入內建主題（固定順序）
    for name, label in built_in.items():
        result.append({
            "name": name,
            "label": label,
            "custom": False,
            "url": f"/static/themes/{name}.css",
            "updated_at": 0.0,
        })

    # 2. 掃描自訂主題（排除內建）
    if themes_dir.exists() and themes_dir.is_dir():
        for item in sorted(themes_dir.iterdir()):
            if item.is_file() and item.suffix.lower() == ".css":
                name = item.stem
                if name not in built_in:
                    stat = item.stat()
                    result.append({
                        "name": name,
                        "label": name,
                        "custom": True,
                        "url": f"/static/themes/{item.name}",
                        "updated_at": stat.st_mtime,
                    })

    return result
```

**設計決策**：
- `sorted(themes_dir.iterdir())` 保證目錄掃描順序穩定（不依賴 OS inode 順序）；前端會按 `updated_at` 排序，但穩定底層有利 debug
- `upload_theme` 的 5 道安全過濾邏輯 100% 未動
- 不引入任何 DB 操作或第三方 API 呼叫，純 IO → 符合極短交易原則

```python
def test_list_themes_endpoint(test_client):
    """RAG-13：GET /api/themes 應回傳 4 個內建主題（custom=False），上傳後追加自訂項目。"""
    client, tmp = test_client

    # 1. 未上傳時：應只有 4 個內建主題，且 custom 均為 False
    response = client.get("/api/themes")
    assert response.status_code == 200
    data = response.json()
    assert len(data) == 4
    for t in data:
        assert t["custom"] is False
        assert t["updated_at"] == 0.0

    # 2. 內建主題順序應固定：mies → kahn → kandinsky → nara
    names = [t["name"] for t in data]
    assert names == ["mies", "kahn", "kandinsky", "nara"]

    # 3. 上傳後：len=5，custom=True，updated_at>0
    client.post("/api/themes/upload", files={"file": ("custom_theme.css", b"...", "text/css")})
    data = client.get("/api/themes").json()
    assert len(data) == 5
    custom_items = [t for t in data if t["custom"]]
    assert len(custom_items) == 1
    assert custom_items[0]["name"] == "custom_theme"
    assert custom_items[0]["updated_at"] > 0.0
```

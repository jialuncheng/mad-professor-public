async def event_stream():
    accumulated, grounding_sources, error_msg = [], [], None
    try:
        for chunk in ai_core.query_stream(...):
            if chunk.get('sentence'): accumulated.append(chunk['sentence'])
            if chunk.get('done') and chunk.get('grounding_sources'):
                grounding_sources = chunk['grounding_sources']
            yield f"data: {json.dumps(chunk, ensure_ascii=False)}\n\n"
            await asyncio.sleep(0)
    except Exception as e:
        # Q4：失敗訊息寫 DB 與前端
        error_msg = f"(生成失敗) {str(e)[:200]}"
        logger.error(...)
        yield "data: " + json.dumps({'sentence': error_msg, 'done': True}, ...) + "\n\n"
    finally:
        full = ''.join(accumulated) if accumulated else (error_msg or '')
        if full.strip():
            try:
                paper_manager.append_chat_message(

                    grounding_sources=grounding_sources or None,

## Q4 失敗訊息處理

baron 決：**寫 DB**（避免「消失感」）。

- `try` 正常 yield + accumulate
- `except`：`error_msg = "(生成失敗) {first 200 chars}"`、yield 給前端、log full traceback
- `finally`：`full = accumulated.join() or error_msg`；strip 後非空才 append；append 本身 try/except（避免 DB 故障擋住 finally）

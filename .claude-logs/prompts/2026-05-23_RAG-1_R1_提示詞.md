```python
# paper_manager.py 新增（位置：set_paper_folder 之後、依現況決定）

def set_paper_tags(session, owner_id: int, paper_uuid: str,
                   tags: list[str]) -> Paper:
    """覆寫 paper 的 user_tags（依 RAG-1 前端執行計劃 §4.1 子項 A）。
    
    用於前端 `#` Modal 編輯 tag 後、PATCH /api/papers/{paper_uuid} 帶 tags=[...]
    走此 helper 整批覆寫。
    
    Args:
        session: SQLAlchemy session
        owner_id: 用戶 ID
        paper_uuid: paper UUID
        tags: 新的 tag list（整批覆寫、非 append）
    
    Returns:
        更新後的 Paper instance
    
    Raises:
        ValueError: paper 不存在
    """
    p = session.query(Paper).filter_by(
        owner_id=owner_id, paper_uuid=paper_uuid
    ).one_or_none()
    if p is None:
        raise ValueError("論文不存在")
    
    # 解析 metadata_json、寫入 user_tags、序列化回去
    meta = json.loads(p.metadata_json) if p.metadata_json else {}
    # 過濾 None / 空字串、strip 空白
    cleaned = [t.strip() for t in (tags or []) if t and t.strip()]
    meta["user_tags"] = cleaned
    p.metadata_json = json.dumps(meta, ensure_ascii=False)
    session.commit()
    return p
```

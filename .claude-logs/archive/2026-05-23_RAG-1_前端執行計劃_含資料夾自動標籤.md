```python
# paper_manager.py 加 helper（~12 行）
def _apply_folder_path_tags(session, owner_id: int, paper_uuid: str,
                             folder_id: Optional[int]) -> None:
    """新需求（自動標籤）：把 folder 路徑名（lowercase）upsert 到 paper 的 user_tags。

    依 .claude-logs/2026-05-23_RAG-1_前端執行計劃_含資料夾自動標籤.md §4.2
    - folder_id=None → 不加（未分類沒有 path tag、Q7）
    - 階層 root → leaf 各層名 lowercase 全部 upsert
    - 既有 user_tags 不刪、只 append + de-dup（Q6）
    - 大小寫一致化只對自動加的 tag、用戶手動加的不動（Q4）
    """
    if folder_id is None:
        return  # 移到未分類、不加（Q7）
    # 取 folder root → leaf path
    path_names = _folder_ancestor_path_names(session, owner_id, folder_id)
    auto_tags = [n.lower() for n in path_names if n]
    if not auto_tags:
        return

    p = session.query(Paper).filter_by(
        owner_id=owner_id, paper_uuid=paper_uuid
    ).one_or_none()
    if p is None:
        return
    meta = json.loads(p.metadata_json) if p.metadata_json else {}
    existing = list(meta.get("user_tags", []))
    # de-dup：用 set 比對 lowercase、但保留既有 tag 原始大小寫（Q4）
    existing_lower = {t.lower() for t in existing}
    for t in auto_tags:
        if t not in existing_lower:
            existing.append(t)
            existing_lower.add(t)
    meta["user_tags"] = existing
    p.metadata_json = json.dumps(meta, ensure_ascii=False)
    session.commit()


def _folder_ancestor_path_names(session, owner_id: int, folder_id: int) -> list[str]:
    """遞迴取得 folder 從 root → leaf 的階層 name 清單。"""
    names = []
    cur_id = folder_id
    while cur_id is not None:
        f = _get_folder_row(session, owner_id, cur_id)
        if f is None:
            break
        names.append(f.name)
        cur_id = f.parent_id
    return list(reversed(names))  # root → leaf
```
